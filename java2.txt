import javafx.application.Application;
import javafx.geometry.Insets;
import javafx.geometry.Pos;
import javafx.scene.Scene;
import javafx.scene.control.*;
import javafx.scene.layout.GridPane;
import javafx.scene.layout.VBox;
import javafx.stage.Stage;

public class TipCalculator extends Application {

    @Override
    public void start(Stage stage) {

        Label title = new Label("Tip Calculator");
        title.setStyle("-fx-font-size: 22px; -fx-font-weight: bold;");

        Label billLabel = new Label("Bill Amount:");
        TextField billField = new TextField();
        billField.setPromptText("Enter bill amount");

        Label tipLabel = new Label("Tip Percentage:");
        ComboBox<Integer> tipBox = new ComboBox<>();
        tipBox.getItems().addAll(5, 10, 15, 20, 25);
        tipBox.setValue(15);

        Label peopleLabel = new Label("Number of People:");
        Spinner<Integer> peopleSpinner =
                new Spinner<>(1, 100, 1);

        CheckBox roundCheckBox =
                new CheckBox("Round tip to nearest rupee");

        Button calculateButton = new Button("Calculate Tip");
        Button clearButton = new Button("Clear");

        Label tipResult = new Label("Tip Amount: ₹0.00");
        Label totalResult = new Label("Total Bill: ₹0.00");
        Label splitResult = new Label("Per Person: ₹0.00");
        Label messageLabel = new Label();

        GridPane grid = new GridPane();
        grid.setHgap(10);
        grid.setVgap(12);

        grid.add(billLabel, 0, 0);
        grid.add(billField, 1, 0);

        grid.add(tipLabel, 0, 1);
        grid.add(tipBox, 1, 1);

        grid.add(peopleLabel, 0, 2);
        grid.add(peopleSpinner, 1, 2);

        grid.add(roundCheckBox, 0, 3, 2, 1);

        // Handle Calculate button click
        calculateButton.setOnAction(e -> {
            try {
                double bill = Double.parseDouble(
                        billField.getText().trim());

                int percentage = tipBox.getValue();
                int people = peopleSpinner.getValue();

                if (bill < 0) {
                    messageLabel.setText(
                            "Enter a valid bill amount.");
                    return;
                }

                double tip = bill * percentage / 100.0;

                if (roundCheckBox.isSelected()) {
                    tip = Math.round(tip);
                }

                double total = bill + tip;
                double perPerson = total / people;

                tipResult.setText(String.format(
                        "Tip Amount: ₹%.2f", tip));

                totalResult.setText(String.format(
                        "Total Bill: ₹%.2f", total));

                splitResult.setText(String.format(
                        "Per Person: ₹%.2f", perPerson));

                messageLabel.setText("Calculation completed.");

            } catch (NumberFormatException ex) {
                messageLabel.setText(
                        "Please enter a valid bill amount.");
            }
        });

        // Handle Clear button click
        clearButton.setOnAction(e -> {
            billField.clear();
            tipBox.setValue(15);
            peopleSpinner.getValueFactory().setValue(1);
            roundCheckBox.setSelected(false);

            tipResult.setText("Tip Amount: ₹0.00");
            totalResult.setText("Total Bill: ₹0.00");
            splitResult.setText("Per Person: ₹0.00");
            messageLabel.setText("");
        });

        VBox root = new VBox(15);
        root.setPadding(new Insets(20));
        root.setAlignment(Pos.CENTER_LEFT);

        root.getChildren().addAll(
                title,
                grid,
                calculateButton,
                clearButton,
                tipResult,
                totalResult,
                splitResult,
                messageLabel
        );

        Scene scene = new Scene(root, 400, 450);

        stage.setTitle("Tip Calculator");
        stage.setScene(scene);
        stage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
